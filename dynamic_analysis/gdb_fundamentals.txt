/
find_sqrt - Finds the natural square root recursively
@n: Number to calculate
@i: Number being tested*
Return: Square root, or -1 if not found*/
int find_sqrt(int n, int i)
{
    if (i == n / i && n % i == 0)
        return (i);

    if (i > n / i)
        return (-1);

    return (find_sqrt(n, i + 1));
}
